\section{Panoramica del sistema}
% ============================================================

Il sistema controlla il braccio \textbf{Unitree Z1} montato su \textbf{Boston Dynamics Spot}
per eseguire autonomamente il protocollo \textbf{FAST ultrasound} su un paziente sdraiato.
Lo stack completo è composto da:

\begin{itemize}
  \item \textbf{Percezione Spot (Orbbec)}: YOLO11 skeleton + classificazione postura +
        rilevamento persona sdraiata + approach point (\texttt{spot\_perception})
  \item \textbf{Percezione Z1 (RealSense)}: YOLO11 torso tracker + Kalman 3D +
        stima piano superficiale PCA (\texttt{z1\_vision})
  \item \textbf{WBC — Whole Body Control} (\texttt{spot\_control}):
        QP olistico Spot+Z1 durante l'approccio; FSM 6 stati; mux priorità arm goal
  \item \textbf{Controllo posizione Z1}: IK Jacobiana pseudo-inversa + JointTrajectoryController
  \item \textbf{Controllo forza Z1}: Impedance controller cartesiano (500 Hz, torque control)
  \item \textbf{Sicurezza}: keyboard safety, stati HOMING/EMERGENCY, workspace clipping
\end{itemize}

\subsection*{Flusso della missione}
\begin{enumerate}
  \item \textbf{Approaching}: WBC QP coordina Spot (cmd\_vel) + Z1 (ik\_goal) simultaneamente;
        braccio pre-punta il paziente mentre Spot cammina
  \item \textbf{Handoff}: Orbbec cede a RealSense; z1\_FSM prende controllo del braccio
  \item \textbf{FAST scan}: z1\_FSM esegue 5 punti anatomici (Hub, Subxiphoid, RUQ, LUQ, Suprapubic)
        con impedance control; Spot fermo
  \item \textbf{WS Extension}: se punto FAST irraggiungibile, WBC torna attivo per
        micro-step Spot + compensazione braccio simultanea
\end{enumerate}

\section{Come avviare il sistema}
% ============================================================

\subsection*{Prerequisiti}
\begin{itemize}[leftmargin=1.5em]
  \item SpotCore con \texttt{spot\_ros2} driver in esecuzione
  \item Jetson con ROS2 Humble, Pinocchio, driver Z1, RealSense SDK, Orbbec SDK, YOLO11
  \item \texttt{ROS\_DOMAIN\_ID} uguale su Jetson e SpotCore
  \item Workspace compilato: \texttt{colcon build --packages-select z1\_vision spot\_perception spot\_control}
\end{itemize}

\subsection*{Sequenza di lancio completa (sistema integrato)}

\begin{lstlisting}
# Su ogni terminale Jetson:
source /opt/ros/humble/setup.bash && source ~/TERESA_ws/install/setup.bash
export ROS_DOMAIN_ID=0
\end{lstlisting}

\begin{center}
\begin{tabular}{|c|l|p{7.5cm}|}
\hline
\textbf{T} & \textbf{Macchina} & \textbf{Comando} \\
\hline
1 & SpotCore & \texttt{ros2 launch spot\_ros2 spot\_driver.launch.py} \\
\hline
2 & Jetson & \texttt{ros2 launch spot\_perception spot\_perception.launch.py} \\
\hline
3 & Jetson & \texttt{ros2 launch z1\_vision z1\_realsense.launch.py use\_rviz:=true} \\
\hline
4 & Jetson & \texttt{ros2 launch z1\_vision z1\_perception.launch.py} \\
\hline
5 & Jetson & \texttt{ros2 launch z1\_vision z1\_control.launch.py fsm\_delay:=5.0} \\
\hline
6 & Jetson & \texttt{ros2 launch spot\_control wbc.launch.py} \\
\hline
7 & Jetson & \texttt{ros2 run z1\_vision z1\_keyboard\_safety} \\
\hline
\end{tabular}
\end{center}

\textbf{Nota}: il terminale 6 (WBC) va avviato dopo T1--T5.
Il \texttt{wbc\_coordinator} parte in IDLE e inizia l'approccio solo
quando \texttt{spot\_perception} rileva una persona sdraiata con confidenza sufficiente.

\subsection*{Parametri utili}

\begin{itemize}[leftmargin=1.5em]
  \item \texttt{fsm\_delay:=5.0} — secondi prima che z1\_FSM inizi (lascia tempo al setup)
  \item \texttt{use\_rviz:=true} — lancia RViz2 con la configurazione del sistema
  \item \texttt{z1\_x:=0.30 z1\_z:=0.70} — offset TF \texttt{body}$\to$\texttt{link00}
        (stimato; misurare dopo montaggio fisico)
\end{itemize}

\subsection*{FSM z1\_FSM.py — stati}

\begin{center}
\begin{tabular}{lp{9.5cm}}
\toprule
\textbf{Stato} & \textbf{Descrizione} \\
\midrule
\texttt{HOMING}              & Stato iniziale. Porta il braccio in home pose via IK/JTC \\
\texttt{WAITING}             & Aspetta target LOCKED da RealSense tracker \\
\texttt{BODY\_SCANNING}      & Scansione a griglia per trovare posizione ottimale del torso \\
\texttt{CHECKING\_WORKSPACE} & Verifica raggiungibilità target (Pinocchio, non bloccante) \\
\texttt{APPROACHING}         & Invia goal IK verso il target \\
\texttt{WAIT\_IK\_DONE}      & Attende \texttt{/ik\_done=True} dal nodo IK \\
\texttt{SWITCHING\_TO\_TORQUE} & Switch JTC $\to$ torque\_controller \\
\texttt{IMPEDANCE\_RUNNING}  & Impedance controller: approach $\to$ hold $\to$ retract \\
\texttt{SWITCHING\_TO\_JTC}  & Switch torque\_controller $\to$ JTC \\
\texttt{SCAN\_PRELIFT}       & Retrazione tra punti FAST \\
\texttt{EMERGENCY}           & Stop bloccante, attende \texttt{reset} da tastiera \\
\texttt{FAULT}               & Errore fatale (switch fallito) \\
\bottomrule
\end{tabular}
\end{center}

\begin{center}
\begin{tabular}{lp{9.5cm}}

\textbf{Stato} & \textbf{Descrizione} \\

\texttt{APPROACHING}  & QP attivo: Spot naviga + Z1 punta il paziente simultaneamente \\

\end{itemize}

\section{Note tecniche}
% ============================================================

\begin{itemize}
  \item \textbf{IK}: Jacobiana pseudo-inversa smorzata, frame LOCAL\_WORLD\_ALIGNED,
        smoothstep quintico (zero velocità a inizio/fine),
        joint unwrap con \texttt{\_make\_target\_near}.
        \texttt{ik\_rot\_weight: 0.8} per EE perpendicolare alla superficie
  \item \textbf{WBC QP}: weighted damped least-squares,
        $w_{\text{arm}} = \lambda_{\text{arm}} / (m + \varepsilon)$,
        $w_{\text{base}} = \lambda_{\text{base}}$;
        alta manipolabilità $\Rightarrow$ braccio preferito, bassa $\Rightarrow$ base contribuisce di più.
        Update ogni 1.5\,s (rispetta \texttt{traj\_min\_time=1.0\,s} di z1\_ik\_to\_jtc).
        \textbf{Yaw task}: Jacobiana estesa $7\times8$ con riga
        $[0,\ldots,0,1]$ per allineare Spot perpendicolare all'asse testa--piedi.
        \textbf{Stochastic dead zone}: velocità nulla quando EE già dentro bolla
        $r = z_\delta \cdot \sigma_{\max}(\mathbf{P}_{\text{pos}})$
        ($z_\delta=1.96$ $\Rightarrow$ 95\,\% confidence, \texttt{z\_delta} in yaml)
  \item \textbf{IK Goal Mux}: z1\_FSM pubblica su \texttt{/z1/ik\_goal\_pose};
        WBC pubblica su \texttt{/wbc/ik\_goal\_pose};
        il mux seleziona in base a \texttt{/wbc/enable}
  \item \textbf{TF body$\to$link00}: offset stimato $x=0.30$\,m, $z=0.70$\,m.
        \textbf{Da misurare fisicamente dopo montaggio Z1 su Spot}
  \item \textbf{URDF path} (hardcoded in yaml): \\
        \texttt{/home/andrea/Ros2\_repositories/unitree\_z1\_ws/install/}\\
        \texttt{z1\_description/share/z1\_description/urdf/z1.urdf}
  \item \textbf{Kalman}: dt adattivo dal timestamp immagine,
        \texttt{vel\_damping = 0.9}
  \item \textbf{Impedance}: 500\,Hz, PID cartesiano + joint-space polso (J4--J6),
        compensazione gravity+coriolis
\end{itemize}

% ----
\subsection*{I. Introduction ($\sim$1.5 pag)}
\begin{itemize}
  \item Motivazione clinica: FAST scan in emergenza, necessità di robot mobile
  \item Gap: WBC esistente assume target deterministico, non gestisce incertezza percettiva
  \item Contributions (lista numerata — T-RO la richiede esplicitamente):
    \begin{enumerate}
      \item Uncertainty-aware WBC: accoppiamento formale Kalman $\to$ QP
      \item Workspace-gradient continuo (vs trigger binario)
      \item Sistema completo validato su Spot+Z1 hardware reale
    \end{enumerate}
\end{itemize}

% ----
\subsection*{II. Related Work ($\sim$1.5 pag)}
\begin{itemize}
  \item Mobile manipulation WBC (Haviland 2022, Khatib, Bayle 2003)
  \item Legged robot manipulation (ANYmal, Spot papers)
  \item Perception-aware control (stochastic MPC, risk-aware planning)
  \item FAST scan robotico (stato arte: arm fisso, base ferma)
\end{itemize}

% ----
\subsection*{III. Problem Formulation ($\sim$0.8 pag)}
\begin{itemize}
  \item Sistema: Spot (2 DOF non-olonomo) + Z1 (6 DOF)
  \item Notazione: TF chain, frames, decision variable
        $\mathbf{x} = [\dot{\mathbf{q}}^\top, v_x, \omega_z]^\top \in \mathbb{R}^8$
  \item Obiettivo formale: tracking EE con incertezza target bayesiana
\end{itemize}

% ----
\subsection*{IV. Method ($\sim$3.5 pag) \textnormal{--- cuore del paper}}

\textbf{A. Holistic Jacobian}
\begin{itemize}
  \item $\mathbf{J}_{\text{base}}$ analitico (vincolo non-olonomo Spot)
  \item $\mathbf{J}_{\text{holistic}} = [\mathbf{J}_{\text{arm}} \mid \mathbf{J}_{\text{base}}] \in \mathbb{R}^{6\times8}$
\end{itemize}

\textbf{B. Standard WBC QP} (baseline di partenza)
\begin{itemize}
  \item Weighted damped least-squares
  \item Manipulability weighting: $w_{\text{arm}} = \lambda_{\text{arm}}/(m+\varepsilon)$
\end{itemize}

\textbf{C. Perception-Uncertainty-Aware WBC} ← \textit{contributo principale}
\begin{itemize}
  \item Kalman3D: modello, propagazione covarianza $\mathbf{P}$
  \item Accoppiamento formale:
        $\mathbf{v}_{\text{des}} = k_p \,\mathbf{P}_{\text{pos}}^{-1}\,\mathbf{e}$
  \item Analisi: $\mathbf{P}\to\infty \Rightarrow$ robot si ferma;
        $\mathbf{P}\to 0 \Rightarrow$ guadagno massimo
  \item \textbf{Proposizione/Lemma}: bounded control action sotto incertezza
        illimitata (dimostrazione formale richiesta da T-RO)
\end{itemize}

\textbf{D. Continuous Workspace-Gradient} ← \textit{contributo secondario}
\begin{itemize}
  \item $w(\mathbf{q})$: distanza dal boundary workspace Z1
  \item Termine addizionale nel QP:
        $\lambda_{\text{ws}}\,\nabla w(\mathbf{q})^\top \dot{\mathbf{q}}$
\end{itemize}

\section{Design degli esperimenti per IEEE T-RO}
% ============================================================

\subsection*{Claim da dimostrare}
\begin{enumerate}
  \item \textbf{C1} --- L'uncertainty-aware WBC riduce l'errore di tracking quando
        la detection è incerta
  \item \textbf{C2} --- Il WBC (base+arm simultanei) riduce il tempo di missione
        rispetto a baseline arm-only
  \item \textbf{C3} --- Il workspace-gradient riduce i fallimenti di missione
        rispetto al trigger binario
\end{enumerate}

\subsection*{Baseline}
\begin{itemize}
  \item \textbf{B0} --- Spot fermo, solo Z1 (approccio classico in letteratura)
  \item \textbf{B1} --- WBC standard, guadagni fissi, nessuna uncertainty
        (sistema attuale)
  \item \textbf{B2} --- WBC uncertainty-aware (metodo proposto)
\end{itemize}

\subsection*{Setup fisico}
\begin{itemize}
  \item Manichino antropomorfico come paziente
  \item AprilTag sul torso per ground truth posizione target
        (via \texttt{apriltag\_ros} + TF esterna)
  \item Ground truth EE: FK con Pinocchio da joint states
        (oppure mocap se disponibile)
  \item Griglia 3$\times$3: distanza $\{1.0, 1.5, 2.0\}$\,m
        $\times$ angolo $\{-30°, 0°, +30°\}$ = 9 configurazioni
  \item 10 trial per configurazione per baseline $\Rightarrow$ 270 trial totali
  \item Logging: \texttt{rosbag} completo, analisi offline Python/matplotlib
  \item Statistiche: t-test o Mann-Whitney tra B1 e B2
\end{itemize}

% ----
\subsection*{E1 — Uncertainty vs Tracking Error \textnormal{(valida C1, cuore del paper)}}

\textbf{Procedura}: avvio missione con visibilità normale ($\mathbf{P}$ piccola),
occlusione parziale del manichino a metà approach ($\mathbf{P}$ sale),
rimozione occlusione ($\mathbf{P}$ scende).

\textbf{Misure}:
\begin{itemize}
  \item $\mathrm{tr}(\mathbf{P}_{\text{pos}}(t))$ nel tempo
  \item $\|\mathbf{p}_{\text{EE}}(t) - \mathbf{p}_{\text{target}}\|$ nel tempo
  \item $\|\mathbf{v}_{\text{cmd}}(t)\|$ (velocità comandata)
\end{itemize}

\textbf{Risultato atteso}: con B1 (guadagni fissi) il robot continua veloce
durante l'occlusione $\Rightarrow$ errore alto. Con B2 il robot rallenta
$\Rightarrow$ errore contenuto. Grafico a 3 curve temporali sincronizzate.

% ----
\subsection*{E2 — Mission Time \textnormal{(valida C2)}}

\textbf{Procedura}: 9 configurazioni $\times$ 10 trial $\times$ 3 baseline = 270 trial.

\textbf{Misure}:
\begin{itemize}
  \item $t_{\text{total}}$: start $\to$ \texttt{/scan\_done}
  \item $t_{\text{approach}}$: start $\to$ HANDOFF
  \item $t_{\text{scan}}$: HANDOFF $\to$ scan done
  \item Percentuale missioni completate con successo
\end{itemize}

\textbf{Risultato atteso}: box plot per distanza (1.0/1.5/2.0\,m) separatamente.
Il vantaggio di B2 su B0 cresce con la distanza.

% ----
\subsection*{E3 — Workspace Extension \textnormal{(valida C3)}}

\textbf{Scenario}: manichino posizionato al boundary del workspace Z1
(trigger binario fallisce spesso, gradient no).

\textbf{Procedura}: 5 posizioni boundary $\times$ 10 trial
$\times$ 2 metodi (trigger binario vs workspace-gradient) = 100 trial.

\textbf{Misure}:
\begin{itemize}
  \item Percentuale trial completati
  \item Numero di WS\_EXTENSION events per trial
  \item Jitter EE durante WS\_EXTENSION:
        $\sigma(\mathbf{p}_{\text{EE}})$
\end{itemize}

\textbf{Confronto}: B0 (Spot fermo = baseline zero jitter) vs B1 vs B2.
