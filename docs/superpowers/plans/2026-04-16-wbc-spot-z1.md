**Goal:** Implement Whole Body Control for Spot+Z1 — during approach the arm continuously compensates Spot's motion to keep RealSense pointed at the patient; during FAST scan, QP computes micro-steps for Spot when Z1 workspace is insufficient.

**Architecture:** A `wbc_qp_controller` node solves a holistic weighted least-squares problem at 1Hz partitioning EE motion between Spot base (cmd_vel) and Z1 arm (ik_goal_pose). A `wbc_coordinator` FSM manages phases (APPROACHING, CONFIRMING, HANDOFF, SCANNING, WS_EXTENSION) and feeds the QP with the correct EE goal. z1_vision is untouched — the QP interfaces via existing /ik_goal_pose and /ik_enable topics.

**Tech Stack:** Python 3, ROS2 Humble, NumPy, Pinocchio (own model instance), tf2_ros

```python
# src/spot_control/spot_control/wbc_math.py
"""
Pure WBC math — no ROS, fully unit-testable.

Conventions:
  Spot body frame: X=forward, Y=left, Z=up
  cmd_vel: [vx, wz]  (non-holonomic)
  Z1 joints: 6 DOF
  Decision variable: x = [q_dot(6), vx, wz]  shape (8,)
"""
import numpy as np


def compute_j_base(p_ee_in_body: np.ndarray) -> np.ndarray:
    """
    Analytical Jacobian of EE w.r.t. Spot base velocity [vx, wz].

    For a non-holonomic base moving at (vx, wz) in body frame,
    the EE velocity (in body frame) is:
      v_lin = [vx, 0, 0] + wz × p_ee
      v_ang = [0, 0, wz]

    Args:
        p_ee_in_body: EE position in Spot body frame, shape (3,)

    Returns:
        J_base: shape (6, 2)  rows = [wx, wy, wz, vx, vy, vz] (Pinocchio local convention)
        Note: Pinocchio spatial velocity is [angular(3), linear(3)]
    """
    px, py, pz = p_ee_in_body

    # Pinocchio spatial velocity convention: [omega(3), v(3)]
    # omega due to base: [0, 0, wz]
    # v due to base:     [vx - wz*py, wz*px, 0]
    J = np.zeros((6, 2))

    # Column 0: vx
    J[3, 0] = 1.0   # linear x

    # Column 1: wz
    J[2, 1] = 1.0          # angular z
    J[3, 1] = -py          # linear x coupling: -py * wz
    J[4, 1] =  px          # linear y coupling:  px * wz

    return J


def compute_j_holistic(J_arm: np.ndarray, J_base: np.ndarray) -> np.ndarray:
    """
    Assemble holistic Jacobian: J = [J_arm | J_base]

    Args:
        J_arm:  shape (6, 6)
        J_base: shape (6, 2)

    Returns:
        J_holistic: shape (6, 8)
    """
    return np.hstack([J_arm, J_base])


def manipulability(J_arm: np.ndarray) -> float:
    """
    Yoshikawa manipulability measure: sqrt(det(J J^T))

    Returns 0.0 if J is singular (det <= 0).
    """
    val = np.linalg.det(J_arm @ J_arm.T)
    return float(np.sqrt(max(val, 0.0)))


def wbc_split(
    J_holistic: np.ndarray,
    v_des: np.ndarray,
    m: float,
    lam_arm: float = 1.0,
    lam_base: float = 1.0,
    damping: float = 1e-3,
    vx_max: float = 0.4,
    wz_max: float = 0.5,
    q_dot_max: float = 0.6,
) -> tuple[np.ndarray, float, float]:
    """
    Weighted damped least-squares split of v_des between arm and base.

    Weight matrix W: diagonal, arm DOFs weighted by lam_arm / (m + eps),
    base DOFs weighted by lam_base. When m is high (arm well-conditioned)
    the arm weight is low → arm absorbs more of the task.
    When m is low (near singular) arm weight increases → base takes over.

    Args:
        J_holistic: shape (6, 8)
        v_des:      desired EE spatial velocity, shape (6,)
        m:          current arm manipulability scalar
        lam_arm:    base weight for arm DOFs
        lam_base:   weight for base DOFs
        damping:    damping factor for regularization
        vx_max:     Spot max linear velocity [m/s]
        wz_max:     Spot max angular velocity [rad/s]
        q_dot_max:  Z1 max joint velocity [rad/s]

    Returns:
        q_dot: arm joint velocities, shape (6,)
        vx:    Spot forward velocity [m/s]
        wz:    Spot yaw velocity [rad/s]
    """
    eps = 1e-4
    arm_weight = lam_arm / (m + eps)   # high m → low arm weight → arm preferred
    base_weight = lam_base

    W = np.diag([arm_weight] * 6 + [base_weight] * 2)

    # Weighted damped least-squares:
    # x = W^{-1} J^T (J W^{-1} J^T + damping*I)^{-1} v_des
    W_inv = np.diag(1.0 / np.diag(W))
    A = J_holistic @ W_inv @ J_holistic.T + damping * np.eye(6)
    x = W_inv @ J_holistic.T @ np.linalg.solve(A, v_des)

    q_dot = np.clip(x[:6], -q_dot_max, q_dot_max)
    vx    = float(np.clip(x[6], -vx_max,  vx_max))
    wz    = float(np.clip(x[7], -wz_max,  wz_max))

    return q_dot, vx, wz
```

```python
#!/usr/bin/env python3
"""
WBC QP Controller
Reads EE goal from /wbc/ee_goal, current arm state from /joint_states
and TF chain, then solves holistic WBC split and publishes:
  - /ik_goal_pose  → z1_ik_to_jtc  (arm share, PoseStamped)
  - /my_spot/cmd_vel → Spot         (base share, Twist)

Update rate: configurable (default 1.5s — respects z1_ik_to_jtc traj_min_time).
Enabled/disabled via /wbc/enable (Bool).
"""
import numpy as np
import pinocchio as pin

import rclpy
from rclpy.node import Node
from rclpy.duration import Duration
import rclpy.time

from geometry_msgs.msg import PoseStamped, Twist
from sensor_msgs.msg import JointState
from std_msgs.msg import Bool

from tf2_ros import Buffer, TransformListener, TransformException
import tf2_geometry_msgs  # noqa: F401

from spot_control.wbc_math import (
    compute_j_base,
    compute_j_holistic,
    manipulability,
    wbc_split,
)


JOINT_ORDER = ['joint1', 'joint2', 'joint3', 'joint4', 'joint5', 'joint6']


class WBCQPControllerNode(Node):

    def __init__(self):
        super().__init__('wbc_qp_controller')

        # ── Parameters ────────────────────────────────────────────────
        self.declare_parameter('urdf_path', '')
        self.declare_parameter('odom_frame',    'my_spot/odom')
        self.declare_parameter('body_frame',    'my_spot/body')
        self.declare_parameter('z1_base_frame', 'link00')
        self.declare_parameter('ee_frame',      'link06')
        self.declare_parameter('lam_arm',       1.0)
        self.declare_parameter('lam_base',      1.0)
        self.declare_parameter('damping',       1e-3)
        self.declare_parameter('kp_pos',        1.0)
        self.declare_parameter('kp_ang',        0.5)
        self.declare_parameter('vx_max',        0.4)
        self.declare_parameter('wz_max',        0.5)
        self.declare_parameter('q_dot_max',     0.6)
        self.declare_parameter('update_period', 1.5)
        self.declare_parameter('ik_goal_topic',       '/ik_goal_pose')
        self.declare_parameter('ik_enable_topic',     '/ik_enable')
        self.declare_parameter('cmd_vel_topic',       '/my_spot/cmd_vel')
        self.declare_parameter('joint_states_topic',  '/joint_states')

        p = lambda n: self.get_parameter(n).value
        self._odom_frame    = p('odom_frame')
        self._body_frame    = p('body_frame')
        self._z1_base_frame = p('z1_base_frame')
        self._ee_frame      = p('ee_frame')
        self._lam_arm       = float(p('lam_arm'))
        self._lam_base      = float(p('lam_base'))
        self._damping       = float(p('damping'))
        self._kp_pos        = float(p('kp_pos'))
        self._kp_ang        = float(p('kp_ang'))
        self._vx_max        = float(p('vx_max'))
        self._wz_max        = float(p('wz_max'))
        self._q_dot_max     = float(p('q_dot_max'))
        self._update_period = float(p('update_period'))

        # ── Pinocchio ─────────────────────────────────────────────────
        urdf = p('urdf_path')
        self._model = pin.buildModelFromUrdf(urdf)
        self._data  = self._model.createData()
        self._ee_id = self._model.getFrameId(self._ee_frame)
        self._q     = pin.neutral(self._model)

        # ── TF ────────────────────────────────────────────────────────
        self._tf = Buffer()
        TransformListener(self._tf, self)

        # ── State ─────────────────────────────────────────────────────
        self._enabled   = False
        self._goal:  PoseStamped | None = None
        self._q_meas: np.ndarray | None = None

        # ── Sub / Pub ─────────────────────────────────────────────────
        self.create_subscription(Bool,        '/wbc/enable',         self._cb_enable,  10)
        self.create_subscription(PoseStamped, '/wbc/ee_goal',        self._cb_goal,    10)
        self.create_subscription(JointState,  p('joint_states_topic'), self._cb_joints, 50)

        self._pub_ik   = self.create_publisher(PoseStamped, p('ik_goal_topic'),   10)
        self._pub_en   = self.create_publisher(Bool,        p('ik_enable_topic'), 10)
        self._pub_vel  = self.create_publisher(Twist,       p('cmd_vel_topic'),   10)

        # ── Timer ─────────────────────────────────────────────────────
        self.create_timer(self._update_period, self._update)
        self.get_logger().info('WBC QP Controller ready.')

    # ── Callbacks ─────────────────────────────────────────────────────

    def _cb_enable(self, msg: Bool) -> None:
        self._enabled = msg.data
        if not self._enabled:
            self._pub_vel.publish(Twist())  # stop Spot
            en = Bool(); en.data = False
            self._pub_en.publish(en)

    def _cb_goal(self, msg: PoseStamped) -> None:
        self._goal = msg

    def _cb_joints(self, msg: JointState) -> None:
        name_to_pos = dict(zip(msg.name, msg.position))
        try:
            self._q_meas = np.array([name_to_pos[j] for j in JOINT_ORDER])
        except KeyError:
            pass

    # ── Main update ───────────────────────────────────────────────────

    def _update(self) -> None:
        if not self._enabled or self._goal is None or self._q_meas is None:
            return

        # 1. Get current EE pose in odom via TF
        try:
            ee_in_odom = self._tf.lookup_transform(
                self._odom_frame, self._ee_frame,
                rclpy.time.Time(), timeout=Duration(seconds=0.1))
        except TransformException as e:
            self.get_logger().warn(f'TF ee→odom failed: {e}', throttle_duration_sec=2.0)
            return

        # 2. Get EE position in body frame (for J_base)
        try:
            ee_in_body = self._tf.lookup_transform(
                self._body_frame, self._ee_frame,
                rclpy.time.Time(), timeout=Duration(seconds=0.1))
        except TransformException as e:
            self.get_logger().warn(f'TF ee→body failed: {e}', throttle_duration_sec=2.0)
            return

        # 3. Transform goal to odom frame
        try:
            goal_odom = self._tf.transform(
                self._goal, self._odom_frame,
                timeout=Duration(seconds=0.1))
        except TransformException as e:
            self.get_logger().warn(f'TF goal→odom failed: {e}', throttle_duration_sec=2.0)
            return

        # 4. Compute EE error → desired spatial velocity (odom frame)
        dp = np.array([
            goal_odom.pose.position.x - ee_in_odom.transform.translation.x,
            goal_odom.pose.position.y - ee_in_odom.transform.translation.y,
            goal_odom.pose.position.z - ee_in_odom.transform.translation.z,
        ])
        # Orientation error via log map (simplified: only position for now)
        v_des = np.zeros(6)
        v_des[3:6] = self._kp_pos * dp   # linear part (Pinocchio: [ang, lin])

        # 5. Compute J_arm from Pinocchio
        q = self._q.copy()
        q[:6] = self._q_meas
        pin.computeJointJacobians(self._model, self._data, q)
        pin.updateFramePlacements(self._model, self._data)
        J_arm = pin.getFrameJacobian(
            self._model, self._data, self._ee_id,
            pin.ReferenceFrame.LOCAL_WORLD_ALIGNED)

        # 6. Compute J_base
        p_ee_body = np.array([
            ee_in_body.transform.translation.x,
            ee_in_body.transform.translation.y,
            ee_in_body.transform.translation.z,
        ])
        J_base = compute_j_base(p_ee_body)
        J_hol  = compute_j_holistic(J_arm, J_base)

        # 7. Manipulability
        m = manipulability(J_arm)

        # 8. WBC split
        q_dot, vx, wz = wbc_split(
            J_hol, v_des, m,
            lam_arm=self._lam_arm, lam_base=self._lam_base,
            damping=self._damping,
            vx_max=self._vx_max, wz_max=self._wz_max,
            q_dot_max=self._q_dot_max,
        )

        # 9. Integrate q_dot → new joint config → FK → new EE goal
        dt = self._update_period
        q_new = q.copy()
        q_new[:6] = np.clip(
            self._q_meas + q_dot * dt,
            self._model.lowerPositionLimit[:6],
            self._model.upperPositionLimit[:6],
        )
        pin.forwardKinematics(self._model, self._data, q_new)
        pin.updateFramePlacements(self._model, self._data)
        T_new = self._data.oMf[self._ee_id]

        # 10. Build and publish new EE goal (in z1_base frame — z1_ik_to_jtc expects it there)
        try:
            T_odom_to_z1base = self._tf.lookup_transform(
                self._z1_base_frame, self._odom_frame,
                rclpy.time.Time(), timeout=Duration(seconds=0.1))
        except TransformException as e:
            self.get_logger().warn(f'TF odom→z1base failed: {e}', throttle_duration_sec=2.0)
            return

        # Express T_new (odom frame) in z1_base frame
        t = T_odom_to_z1base.transform.translation
        R_odom_to_z1base = _quat_to_rot(T_odom_to_z1base.transform.rotation)
        p_in_z1base = R_odom_to_z1base @ T_new.translation + np.array([t.x, t.y, t.z])
        R_in_z1base = R_odom_to_z1base @ T_new.rotation

        goal_msg = PoseStamped()
        goal_msg.header.stamp    = self.get_clock().now().to_msg()
        goal_msg.header.frame_id = self._z1_base_frame
        goal_msg.pose.position.x = float(p_in_z1base[0])
        goal_msg.pose.position.y = float(p_in_z1base[1])
        goal_msg.pose.position.z = float(p_in_z1base[2])
        q_out = _rot_to_quat(R_in_z1base)
        goal_msg.pose.orientation.x = float(q_out[0])
        goal_msg.pose.orientation.y = float(q_out[1])
        goal_msg.pose.orientation.z = float(q_out[2])
        goal_msg.pose.orientation.w = float(q_out[3])

        en = Bool(); en.data = True
        self._pub_en.publish(en)
        self._pub_ik.publish(goal_msg)

        # 11. Publish cmd_vel for Spot
        twist = Twist()
        twist.linear.x  = float(vx)
        twist.angular.z = float(wz)
        self._pub_vel.publish(twist)

        self.get_logger().info(
            f'WBC: m={m:.3f} vx={vx:.3f} wz={wz:.3f} |dp|={np.linalg.norm(dp):.3f}',
            throttle_duration_sec=2.0)


# ── Helpers ───────────────────────────────────────────────────────────────────

def _quat_to_rot(q) -> np.ndarray:
    """geometry_msgs Quaternion → 3x3 rotation matrix."""
    from tf_transformations import quaternion_matrix
    R = quaternion_matrix([q.x, q.y, q.z, q.w])
    return R[:3, :3]


def _rot_to_quat(R: np.ndarray) -> np.ndarray:
    """3x3 rotation matrix → [x, y, z, w] quaternion."""
    from tf_transformations import quaternion_from_matrix
    M = np.eye(4)
    M[:3, :3] = R
    return quaternion_from_matrix(M)


def main(args=None):
    rclpy.init(args=args)
    node = WBCQPControllerNode()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass
    finally:
        node.destroy_node()
        rclpy.shutdown()
```

```python
#!/usr/bin/env python3
"""
WBC Coordinator — phase FSM

States:
  IDLE           waiting for LYING detection
  APPROACHING    Spot navigates + Z1 look-at via WBC QP
  CONFIRMING     arm moves to confirm Orbbec detection (better view angle)
  HANDOFF        Orbbec lost → switch to RealSense
  SCANNING       z1_FSM active, WBC QP dormant, Spot stopped
  WS_EXTENSION   z1_FSM requested workspace help → QP micro-step

Transitions:
  IDLE → APPROACHING        when posture=LYING, confidence >= threshold
  APPROACHING → CONFIRMING  when confidence < threshold (Orbbec unsure)
  CONFIRMING → APPROACHING  when confidence >= threshold again
  APPROACHING → HANDOFF     when confidence < threshold AND Spot near patient
  HANDOFF → SCANNING        when z1_FSM active (teresa_mission sent READY_FOR_SCAN)
  SCANNING → WS_EXTENSION   when /wbc/ws_request received from z1_FSM
  WS_EXTENSION → SCANNING   when /ik_done received (micro-step complete)
  any → IDLE                when posture != LYING for > timeout
"""
import rclpy
from rclpy.node import Node
from rclpy.duration import Duration
import rclpy.time

from geometry_msgs.msg import PoseStamped
from std_msgs.msg import Bool, String, Float32
from tf2_ros import Buffer, TransformListener, TransformException
import tf2_geometry_msgs  # noqa: F401

import math


class CoordState:
    IDLE         = 'IDLE'
    APPROACHING  = 'APPROACHING'
    CONFIRMING   = 'CONFIRMING'
    HANDOFF      = 'HANDOFF'
    SCANNING     = 'SCANNING'
    WS_EXTENSION = 'WS_EXTENSION'


class WBCCoordinatorNode(Node):

    def __init__(self):
        super().__init__('wbc_coordinator')

        # ── Parameters ────────────────────────────────────────────────
        self.declare_parameter('orbbec_confidence_threshold', 0.4)
        self.declare_parameter('approach_done_distance',      0.8)
        self.declare_parameter('odom_frame',                  'my_spot/odom')
        self.declare_parameter('body_frame',                  'my_spot/body')
        self.declare_parameter('posture_confidence_topic',    '/human_pose/posture_confidence')
        self.declare_parameter('approach_point_topic',        '/laying_human/approach_point')
        self.declare_parameter('z1_fsm_state_topic',          '/z1_fsm/state')
        self.declare_parameter('wbc_goal_topic',              '/wbc/ee_goal')
        self.declare_parameter('wbc_enable_topic',            '/wbc/enable')
        self.declare_parameter('lying_timeout',               3.0)

        p = lambda n: self.get_parameter(n).value
        self._conf_thr      = float(p('orbbec_confidence_threshold'))
        self._approach_dist = float(p('approach_done_distance'))
        self._odom_frame    = p('odom_frame')
        self._body_frame    = p('body_frame')
        self._lying_timeout = float(p('lying_timeout'))

        # ── TF ────────────────────────────────────────────────────────
        self._tf = Buffer()
        TransformListener(self._tf, self)

        # ── State ─────────────────────────────────────────────────────
        self._state           = CoordState.IDLE
        self._posture         = 'UNKNOWN'
        self._confidence      = 0.0
        self._approach_point: PoseStamped | None = None
        self._last_lying_time = None

        # ── Sub / Pub ─────────────────────────────────────────────────
        self.create_subscription(String,      '/human_pose/posture',
                                 self._cb_posture,  10)
        self.create_subscription(Float32,     p('posture_confidence_topic'),
                                 self._cb_conf,     10)
        self.create_subscription(PoseStamped, p('approach_point_topic'),
                                 self._cb_approach, 10)
        self.create_subscription(String,      p('z1_fsm_state_topic'),
                                 self._cb_z1_state, 10)
        self.create_subscription(Bool,        '/ik_done',
                                 self._cb_ik_done,  10)
        self.create_subscription(Bool,        '/wbc/ws_request',
                                 self._cb_ws_req,   10)

        self._pub_goal   = self.create_publisher(PoseStamped, p('wbc_goal_topic'),  10)
        self._pub_enable = self.create_publisher(Bool,        p('wbc_enable_topic'), 10)
        self._pub_state  = self.create_publisher(String,      '/wbc/state',          10)

        self.create_timer(0.2, self._tick)   # 5Hz FSM
        self.get_logger().info('WBC Coordinator ready.')

    # ── Callbacks ─────────────────────────────────────────────────────

    def _cb_posture(self, msg: String) -> None:
        self._posture = msg.data
        if msg.data == 'LYING':
            self._last_lying_time = self.get_clock().now()

    def _cb_conf(self, msg: Float32) -> None:
        self._confidence = float(msg.data)

    def _cb_approach(self, msg: PoseStamped) -> None:
        self._approach_point = msg

    def _cb_z1_state(self, msg: String) -> None:
        if self._state == CoordState.HANDOFF and 'APPROACHING' in msg.data:
            # z1_FSM started — enter SCANNING
            self._set_state(CoordState.SCANNING)
            self._set_wbc_enabled(False)

    def _cb_ik_done(self, msg: Bool) -> None:
        if self._state == CoordState.WS_EXTENSION and msg.data:
            self._set_state(CoordState.SCANNING)

    def _cb_ws_req(self, msg: Bool) -> None:
        if self._state == CoordState.SCANNING and msg.data:
            self._set_state(CoordState.WS_EXTENSION)

    # ── FSM tick ──────────────────────────────────────────────────────

    def _tick(self) -> None:
        self._check_lying_timeout()

        if self._state == CoordState.IDLE:
            self._tick_idle()
        elif self._state == CoordState.APPROACHING:
            self._tick_approaching()
        elif self._state == CoordState.CONFIRMING:
            self._tick_confirming()
        elif self._state == CoordState.WS_EXTENSION:
            self._tick_ws_extension()

        # Publish FSM state
        s = String(); s.data = self._state
        self._pub_state.publish(s)

    def _check_lying_timeout(self) -> None:
        if self._state == CoordState.IDLE:
            return
        if self._posture != 'LYING' and self._last_lying_time is not None:
            elapsed = (self.get_clock().now() - self._last_lying_time).nanoseconds * 1e-9
            if elapsed > self._lying_timeout:
                self.get_logger().warn('LYING timeout → IDLE')
                self._set_state(CoordState.IDLE)
                self._set_wbc_enabled(False)

    def _tick_idle(self) -> None:
        if (self._posture == 'LYING'
                and self._confidence >= self._conf_thr
                and self._approach_point is not None):
            self._set_state(CoordState.APPROACHING)
            self._set_wbc_enabled(True)

    def _tick_approaching(self) -> None:
        if self._approach_point is None:
            return

        # Publish goal to WBC QP
        self._pub_goal.publish(self._approach_point)

        # Check if confidence dropped → CONFIRMING or HANDOFF
        if self._confidence < self._conf_thr:
            dist = self._distance_to_patient()
            if dist is not None and dist < self._approach_dist * 2:
                self._set_state(CoordState.HANDOFF)
            else:
                self._set_state(CoordState.CONFIRMING)

    def _tick_confirming(self) -> None:
        # Arm is pointing at last known position to recover Orbbec view.
        # If confidence recovers → back to APPROACHING.
        if self._confidence >= self._conf_thr:
            self._set_state(CoordState.APPROACHING)
            return
        if self._approach_point is not None:
            self._pub_goal.publish(self._approach_point)

    def _tick_ws_extension(self) -> None:
        # WBC QP computes micro-step; arm goal is the next FAST point
        # (published separately by z1_FSM via /wbc/ee_goal directly).
        pass  # handled by callbacks (_cb_ik_done)

    # ── Helpers ───────────────────────────────────────────────────────

    def _distance_to_patient(self) -> float | None:
        if self._approach_point is None:
            return None
        try:
            t = self._tf.lookup_transform(
                self._body_frame,
                self._approach_point.header.frame_id,
                rclpy.time.Time(), timeout=Duration(seconds=0.1))
            dx = self._approach_point.pose.position.x + t.transform.translation.x
            dy = self._approach_point.pose.position.y + t.transform.translation.y
            return math.hypot(dx, dy)
        except TransformException:
            return None

    def _set_state(self, new_state: str) -> None:
        if new_state != self._state:
            self.get_logger().info(f'WBC FSM: {self._state} → {new_state}')
            self._state = new_state

    def _set_wbc_enabled(self, enabled: bool) -> None:
        msg = Bool(); msg.data = enabled
        self._pub_enable.publish(msg)


def main(args=None):
    rclpy.init(args=args)
    node = WBCCoordinatorNode()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass
    finally:
        node.destroy_node()
        rclpy.shutdown()
```
