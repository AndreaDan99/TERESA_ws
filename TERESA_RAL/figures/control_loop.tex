% ============================================================
%  CONTROL LOOP  –  WBC + Impedance parallel paths
% ============================================================
\begin{tikzpicture}[
    >=stealth,
    arr/.style  = {->, thick},
    box/.style  = {rectangle, draw, rounded corners=3pt,
                   minimum width=2.2cm, minimum height=0.85cm,
                   font=\footnotesize, align=center, fill=white},
    wbc/.style  = {box, fill=blue!12,  draw=blue!60},
    imp/.style  = {box, fill=green!12, draw=green!60!black},
    rob/.style  = {box, fill=gray!15,  draw=gray!60},
    sw/.style   = {rectangle, draw=purple!70, fill=purple!12,
                   rounded corners=3pt, font=\footnotesize, align=center,
                   minimum width=0.9cm, minimum height=2.6cm},
    sum/.style  = {circle, draw, thick, minimum size=0.65cm,
                   inner sep=0pt, font=\footnotesize},
  ]

  % ── Columns x: -0.2  1.2  3.8  6.4  9.0  11.5 ──────────────
  % ── Rows  y: top=1.5  bot=-1.5  fb=-3.2 ────────────────────

  % INPUT
  \node (ref) at (-0.2, 0) {$\mathbf{p}^*$};

  % SUM NODES
  \node[sum] (e1) at (1.2,  1.5) {};
  \node[sum] (e2) at (1.2, -1.5) {};

  % WBC / IMP / SW / ROBOT / FK
  \node[wbc] (vdes) at (3.8,  1.5) {Vel.\ ref.\\$k_p\,\mathbf{e}_p$};
  \node[wbc] (jh)   at (6.4,  1.5) {$\mathbf{J}_h$ QP\\$\dot{\mathbf{q}}^*$};
  \node[imp] (pid)  at (3.8, -1.5) {PID\\$\mathbf{F}_p$};
  \node[imp] (jtc)  at (6.4, -1.5) {$\mathbf{J}^\top$ + RNEA\\$\boldsymbol{\tau}$};
  \node[sw]  (sw)   at (9.0,  0)   {Safe\\Switch};
  \node[rob] (robot) at (11.5, 0)  {Spot $+$ Z1};
  \node[rob] (fk) at (6.4, -3.2) {FK (Pinocchio)};

  % ── ref → sums ───────────────────────────────────────────────
  \draw[arr] (ref.north) |- (e1.west);
  \draw[arr] (ref.south) |- (e2.west);

  % ── WBC path ─────────────────────────────────────────────────
  \draw[arr] (e1) -- (vdes);
  \draw[arr] (vdes) -- node[above, font=\footnotesize]{$\dot{\mathbf{p}}^*$} (jh);

  % ── jh → sw ──────────────────────────────────────────────────
  \draw[arr] (jh.east)
             -- ++(1.2,0)
             node[above, pos=0.7, font=\footnotesize]{$\mathbf{u}^*$}
             [sharp corners] |- (sw.north);

  % ── Impedance path ───────────────────────────────────────────
  \draw[arr] (e2) -- (pid);
  \draw[arr] (pid) -- node[above, font=\footnotesize]{$\mathbf{F}_p$} (jtc);

  % ── jtc → sw ─────────────────────────────────────────────────
  \draw[arr] (jtc.east)
             -- ++(1.2,0)
             node[below, pos=0.7, font=\footnotesize]{$\boldsymbol{\tau}$}
             [sharp corners] |- (sw.south);

  % ── switch → robot / robot → FK ──────────────────────────────
  \draw[arr] (sw.east) -- (robot.west);
  \draw[arr] (robot.south) -- ++(0,-0.85)
             node[right, font=\footnotesize]{$\mathbf{q},\dot{\mathbf{q}}$}
             -| (fk.east);

  % ── FK → Jacobians ───────────────────────────────────────────
  \draw[arr] (fk.north) -- (jtc.south)
             node[midway, right, font=\footnotesize]{$\mathbf{q},\dot{\mathbf{q}}$};
  \draw[arr] (fk.north) -- ++(0,0.6) -| (jh.south)
             node[near end, right, font=\footnotesize]{$\mathbf{q}$};

  % ── FK → sums (feedback) ─────────────────────────────────────
  \coordinate (fbL) at (-0.4, -3.2);
  \draw[arr] (fk.west) -- (fbL)
             node[midway, below, font=\footnotesize]{$\mathbf{p}_{EE}$};
  \draw[arr] (fbL) |- (e1.south);
  \draw[arr] (fbL) |- (e2.south);

  % ── + labels on sum nodes ────────────────────────────────────
  \node[left=-2pt, font=\footnotesize] at (e1.center) {$+$};
  \node[left=-2pt, font=\footnotesize] at (e2.center) {$+$};

\end{tikzpicture}
