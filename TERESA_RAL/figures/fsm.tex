\begin{tikzpicture}[
    node distance = 1.4cm and 1.7cm,
    auto,
    >=stealth,
    state/.style   = {rectangle, rounded corners=3pt, draw, minimum width=2.6cm,
                      minimum height=1cm, font=\footnotesize,
                      fill=white, thick, inner sep=6pt},
    search/.style  = {state, draw=orange!70, fill=orange!8},
    lock/.style    = {state, draw=red!60, fill=red!8},
    approach/.style = {state, draw=blue!70, fill=blue!8},
    expose/.style  = {state, draw=cyan!65, fill=cyan!8},
    review/.style  = {state, draw=teal!60, fill=teal!8},
    scan/.style    = {state, draw=green!60!black, fill=green!8},
    idle/.style    = {state, draw=gray!70, fill=gray!8},
    wait/.style    = {state, draw=violet!60, fill=violet!8, densely dashed},
    trans/.style   = {->, thick, font=\footnotesize},
    transd/.style  = {->, thick, dashed, font=\footnotesize},
    every label/.style = {font=\footnotesize}
  ]

  % ── Column 1: Init (x≈0.0) ──────────────────────────────────
  \node[idle]  (wtf)                                  {Waiting TF};
  \node[idle]  (idl)  [below=of wtf]                   {Idle};

  % ── Column 2: Search / Lock (x≈1.8) ─────────────────────────
  \node[search] (sr)  [right=1.8cm of idl]                    {Searching};
  \node[lock]   (sl)  [above=of sr]                            {Semi-locking};
  \node[lock]   (lk)  [right=of sl]                            {Locking};

  % ── Column 3: Approach + Exposure (x≈5.0) ──────────────────
  \node[approach] (pa) [right=1.4cm of lk]                     {Pre-approach};
  \node[approach] (ap) [below=of pa]                            {Approaching};
  \node[wait]     (we) [below=1.5cm of ap]                        {Wait. Exp.};
  \node[expose]   (es) [right=1.4cm of ap]                      {Exp. Scanning};
  \node[review]   (er) [below=of es]                            {Exp. Review};

  % ── Column 4: FAST (x≈8.6) ─────────────────────────────────
  \node[scan]    (sc) [right=1.4cm of es]                      {Scanning (FAST)};
  \node[wait]    (wf) [above=of sc]                            {Wait. FAST};
  \node[approach] (ws) [right=1.4cm of sc]                     {WS ext};

  % ── Transitions ─────────────────────────────────────────────

  % Init
  \draw[trans] (wtf) -- node[right]{\footnotesize TF ready} (idl);
  \draw[trans] (idl) -- node[above]{\footnotesize start} (sr);

  % Search / Lock
  \draw[trans] (sr)  -- node[left]{\footnotesize RS guided} (sl);
  \draw[trans] (sl)  -- node[above]{\footnotesize Orbbec} (lk);
  \draw[transd] (sl.west) -- ++(-0.5,0) |- node[pos=0.1,left]{\footnotesize timeout} (sr.west);
  \draw[trans] (sr)  -| node[above left,pos=0.7]{\footnotesize LYING} (lk);
  \draw[transd] (lk.south west) -- node[above left,pos=0.4,yshift=2pt]{\footnotesize timeout} (sr.north east);

  % Lock → Pre-approach
  \draw[trans] (lk)  -- node[above]{\footnotesize samples+ik} (pa);

  % Pre-approach → Approaching
  \draw[trans] (pa)  -- node[left]{\footnotesize EST/LOCKED} (ap);
  \draw[transd] (pa) to[loop right] node[right]{\footnotesize timeout} (pa);

  % Approaching → Exposure (conditional gate)
  \draw[trans] (ap.east) -- node[above,pos=0.45]{\footnotesize handoff} (es.west);
  \draw[transd] (ap) -- node[left]{\footnotesize man.\ gate} (we);

  % Waiting Exposure → Exposure Scanning
  \draw[trans] (we) -- node[above, pos=0.5, xshift = -8pt]{\footnotesize confirm} (es);

  % Exposure Scanning → Exposure Review
  \draw[trans] (es) -- node[right,near start]{\footnotesize /exp.\_ready} (er);

  % Exposure Review → FAST (conditional gate)
  \draw[trans] ([yshift=-0.2cm]er.east) -| node[above,pos=0.25]{\footnotesize terminate} ([xshift=-0.2cm]sc.south);
  \draw[transd] ([yshift=0.2cm]er.east) -- ++(0.8,0) |- node[left,pos=0.4]{\footnotesize man.\ gate} (wf.west);

  % Waiting FAST → Scanning
  \draw[trans] (wf.south) -- node[below right,pos=0.7]{\footnotesize confirm} (sc.north);

  % Scanning ↔ WS ext
  \draw[trans] (sc) -- node[above]{\footnotesize unreach.} (ws);
  \draw[trans] (ws) -- node[below]{\footnotesize ik\_done} (sc);

  % Complete → Idle (passes below all states)
  \draw[transd] ([xshift=+0.2cm]sc.south) -- ++(0,-2.8) -|
    node[pos=0.22,below]{\footnotesize complete} (idl.south);

  % ── Legend removed per reviewer feedback ────────────────────

\end{tikzpicture}
